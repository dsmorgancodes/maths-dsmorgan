\section{Statement}

\begin{theorem}[Euler's identity]\label{thm:euler}
  \[
    e^{i\pi} + 1 = 0.
  \]
\end{theorem}

\section{Setting}

Everything is built from power series, so no geometric intuition about angles
is assumed. \Cref{sec:group} gives a second proof using group theory.

\begin{definition}[Exponential, sine and cosine]\label{def:exp}
  For $z \in \C$, define
  \begin{equation}\label{eq:series}
    \exp(z) = \sum_{n=0}^{\infty} \frac{z^n}{n!}, \qquad
    \cos(z) = \sum_{n=0}^{\infty} \frac{(-1)^n z^{2n}}{(2n)!}, \qquad
    \sin(z) = \sum_{n=0}^{\infty} \frac{(-1)^n z^{2n+1}}{(2n+1)!},
  \end{equation}
  and write $e^{z} = \exp(z)$.
\end{definition}

\begin{definition}[The number $\pi$]\label{def:pi}
  $\pi$ is twice the smallest positive real zero of $\cos$.
\end{definition}

\section{Idea of the proof}

Substitute $i\theta$ into the series for $\exp$. The powers of $i$ cycle through
$1, i, -1, -i$, so the even terms rebuild $\cos\theta$ and the odd terms rebuild
$i \sin\theta$. Then evaluate at $\theta = \pi$.

\section{Proof}

\begin{lemma}[Absolute convergence]\label{lem:converge}
  Each series in \eqref{eq:series} converges absolutely for every $z \in \C$.
\end{lemma}

\begin{proof}
  For $\exp$, the ratio of consecutive terms of $\sum \abs{z}^n / n!$ is
  $\abs{z} / (n+1) \to 0$, so the series converges by the ratio test. The
  series for $\cos$ and $\sin$ are bounded term by term by this one.
\end{proof}

\begin{lemma}[Euler's formula]\label{lem:formula}
  For every $\theta \in \R$, $e^{i\theta} = \cos\theta + i \sin\theta$.
\end{lemma}

\begin{proof}
  Since $i^{2k} = (-1)^k$ and $i^{2k+1} = (-1)^k i$, by \cref{lem:converge} we
  may split the series into even and odd terms:
  \begin{align*}
    e^{i\theta}
    &= \sum_{k=0}^{\infty} \frac{(i\theta)^{2k}}{(2k)!}
     + \sum_{k=0}^{\infty} \frac{(i\theta)^{2k+1}}{(2k+1)!} \\
    &= \sum_{k=0}^{\infty} \frac{(-1)^k \theta^{2k}}{(2k)!}
     + i \sum_{k=0}^{\infty} \frac{(-1)^k \theta^{2k+1}}{(2k+1)!}
     = \cos\theta + i \sin\theta. \qedhere
  \end{align*}
\end{proof}

\begin{lemma}[Values at $\pi$]\label{lem:values}
  $\cos\pi = -1$ and $\sin\pi = 0$.
\end{lemma}

\begin{proof}
  Differentiating \eqref{eq:series} term by term gives $\sin' = \cos$ and
  $\cos' = -\sin$, so $\frac{\dd}{\dd x}(\sin^2 x + \cos^2 x) = 0$ and
  $\sin^2 + \cos^2 = 1$.

  By \cref{def:pi}, $\cos(\pi/2) = 0$, so $\sin(\pi/2) = \pm 1$. Since
  $\sin' = \cos > 0$ on $[0, \pi/2)$ and $\sin 0 = 0$, $\sin(\pi/2) = 1$. The
  addition formulas, which follow from $e^{a+b} = e^a e^b$ and
  \cref{lem:formula}, then give
  \[
    \cos\pi = \cos^2\tfrac{\pi}{2} - \sin^2\tfrac{\pi}{2} = -1, \qquad
    \sin\pi = 2 \sin\tfrac{\pi}{2} \cos\tfrac{\pi}{2} = 0. \qedhere
  \]
\end{proof}

\begin{proof}[Proof of \cref{thm:euler}]
  By \cref{lem:formula,lem:values},
  \[
    e^{i\pi} = \cos\pi + i \sin\pi = -1,
  \]
  so $e^{i\pi} + 1 = 0$.
\end{proof}

\section{A second proof: the circle group}\label{sec:group}

The first proof computes $\cos\pi$ and $\sin\pi$ through the addition formulas. The
second avoids them: it needs only the value at $\pi/2$ and the fact that $\exp$ turns
addition into multiplication.

\begin{lemma}[Homomorphism]\label{lem:hom}
  $\exp(z + w) = \exp(z)\exp(w)$ for all $z, w \in \C$, and $\exp(0) = 1$.
\end{lemma}

\begin{proof}
  The series for $\exp(z)$ and $\exp(w)$ converge absolutely by \cref{lem:converge},
  so their Cauchy product converges to $\exp(z)\exp(w)$. Its $n$th term is
  \[
    \sum_{k=0}^{n} \frac{z^k}{k!} \frac{w^{n-k}}{(n-k)!}
    = \frac{1}{n!} \sum_{k=0}^{n} \binom{n}{k} z^k w^{n-k}
    = \frac{(z + w)^n}{n!}
  \]
  by the binomial theorem, which is the $n$th term of $\exp(z + w)$.
\end{proof}

Let $\mathbb{T} = \set{w \in \C : \abs{w} = 1}$. It is a group under multiplication,
called the \emph{circle group}.

\begin{proposition}[Into the circle]\label{prop:circle}
  The map $E(\theta) = e^{i\theta}$ is a group homomorphism
  $E \colon (\R, +) \to (\mathbb{T}, \cdot)$.
\end{proposition}

\begin{proof}
  The series for $\exp$ has real coefficients, so
  $\overline{e^{i\theta}} = e^{-i\theta}$. By \cref{lem:hom},
  \[
    \abs{e^{i\theta}}^2 = e^{i\theta}\, e^{-i\theta} = e^{0} = 1,
  \]
  so $E$ takes values in $\mathbb{T}$. It is a homomorphism by \cref{lem:hom}.
\end{proof}

\begin{lemma}[A quarter turn]\label{lem:quarter}
  $E(\pi/2) = i$.
\end{lemma}

\begin{proof}
  By \cref{lem:formula} and \cref{def:pi},
  $E(\pi/2) = \cos\frac\pi2 + i \sin\frac\pi2 = i \sin\frac\pi2$. This has modulus
  $1$ by \cref{prop:circle}, so $\sin\frac\pi2 = \pm 1$. On $[0, \pi/2)$ we have
  $\sin' = \cos > 0$ and $\sin 0 = 0$, so $\sin\frac\pi2 > 0$ and $E(\pi/2) = i$.
\end{proof}

\begin{proof}[Second proof of \cref{thm:euler}]
  $E$ is a homomorphism, so
  \[
    e^{i\pi} = E\bigl(\tfrac\pi2 + \tfrac\pi2\bigr) = E\bigl(\tfrac\pi2\bigr)^2
    = i^2 = -1. \qedhere
  \]
\end{proof}

\begin{remark}[The group-theoretic meaning]
  More is true \cite[Prologue]{rudin-rca}: $E$ is surjective with kernel $2\pi\Z$,
  so it induces an isomorphism of groups
  \[
    \R / 2\pi\Z \;\cong\; \mathbb{T}, \qquad \theta + 2\pi\Z \mapsto e^{i\theta}.
  \]
  Each side has exactly one element of order $2$: the class of $\pi$ on the left, and
  $-1$ on the right, since $w^2 = 1$ forces $w = \pm 1$. An isomorphism must match
  them. In this language, Euler's identity says that half a turn is $-1$.
\end{remark}

\section{Remarks}

\begin{remark}
  The identity links five constants ($0$, $1$, $e$, $i$, $\pi$) using
  addition, multiplication and exponentiation exactly once each. The proof
  shows it is less mysterious than it looks: it is \cref{lem:formula} at a
  single point.
\end{remark}

\begin{remark}
  Two facts are used without proof here: that power series can be
  differentiated term by term inside their radius of convergence, and that
  $\cos$ has a positive zero (true since $\cos 0 = 1$ and $\cos 2 < 0$ from
  the alternating series).
\end{remark}
